# in this program we will print the factorial of any number entered by user
import logging

logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s - %(levelname)s - %(message)s',
    datefmt='%Y-%m-%d %H:%M:%S'
)

def fact(n):
    logging.debug('Start of factorial(' + str(n) + ')')
    result = 1
    for i in range(n + 1):
        logging.debug('i is ' + str(i) + ', total is ' + str(result))
        result *= i
    return result

print(fact(5))
