#A short guess program
def Guess_number(no):
    guess = 100
    win = False
    if no > guess:
        print('guess is too high')
    if no < guess:
        print('guess is too low')
    if no == guess:
        print('congrates you won')
        win = True
        return win
    return win

Continue = False
while Continue == False:
    no = int(input('Enter any number'))
    Continue = Guess_number(no)
