first_nth = list[0:n]

# now replacing it with each other
for i in range(n):
    list[i] = last_nth[i]

#Merge two sorted lists into one sorted list (manually).
list1 = [2,5,7,9,1,0,8,23]
list2 = [77,71,70,44,22,711,999]
#sorting list 1
for i in range(len(list1)):
    j = i -1
    comparison = list1[i]
    while comparison < list1[j] and j>=0:
        list1[j+1] = list1[j]
        j-=1
    list1[j+1] = comparison
print(list1)
# sorting list2
for i in range(len(list2)):
    j = i -1
    comparison = list2[i]
    while comparison < list2[j] and j>=0:
        list2[j+1] = list2[j]
        j-=1
    list2[j+1] = comparison
print(list2)
#merging both lists into list3
list3=[]
list3 = list1
list3.extend(list2)
print(list3)

#Password Strength Checker: Check for uppercase, lowercase, digits, special characters using lists.

uppercase = [chr(a) for a in range(65,91) ]
lowercase = [chr(b) for b in range(97,123)]
numbers = [chr(l) for l in range(49,58)]
special_character = [chr(c) for c in range(33,48)]
special_character.extend([chr(d) for d in range(58,65)])
special_character.extend([chr(e) for e in range(91,97)] )
special_character.extend([chr(f) for f in range(123,127)])
while True:
    password = input('enter your password ')
    if len(password) < 8:
        print('the length of password should be atleast 8')
    elif not any(ch in password for ch in special_character):
        print('password must hold special_character\n')
    elif not any(ch in password for ch in uppercase):
        print('passcode must contain uppercase letters')
    elif not any(ch in password for ch in lowercase):
        print('passcode must contain lowercase letters')
    elif not any(ch in password for ch in numbers):
        print('it must also hold numbers(0-9)')
    else:
        print('you got strong password!! ')
        exit()

#Simple Encryption: Shift each character’s ASCII value by +1 and store in a list.
simple_text = input("Enter text to encode\n>>")
encoded = []
simple_text = simple_text.lower()
for i in range(len(simple_text)):
    print(f"ASCII valuse of {simple_text[i]} is: {ord(simple_text[i])}")
    ascii = (ord(simple_text[i]) + 1) % 26 + 97
    encoded.append(chr(ascii))
print(''.join(encoded))
